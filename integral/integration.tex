    \section{Integration of Differential Forms}
    \subsection{Partition of Unity}
    \begin{definition}[Support]\label{def:support}
        Let \(X\) be a topological space, and \(f : X \to \mathbb{R}\). Then 
        the support of \(f\) is defined as
        \[
        \supp f := \Set{x \in X|f(x) \neq 0}.
        \]
    \end{definition}
    \begin{theorem}[Partition of Unity]\label{thm:manifold-partition-of-unity}
        Let \(\manifold[M]\) be a \(C^\infty\) manifold with dimension \(m\) with 
        atlas \(\Phi\). \\
        Let \(\Phi = \Set{\phi_j|\phi_j : V_j \to \phi_j(V_j), j \in J}\).\\
        Then it is possible to construct a set of \(C^\infty\) functions 
        \(\rho_j, j \in J\) s.t. 
        \[
        1 = \sum_{j \in J}^{} \rho_j,\; \supp \rho_j \subseteq V_j
        \]
    \end{theorem}
    \subsection{Orientation}
    \subsubsection{Definition}
    \begin{definition}[Compatible Coordinate Charts]\label{def:compatible-coord-charts}
        Given a manifold \(\manifold[M]\), its two coordinate charts are called 
        compatible (have the same orientation) if, 
        \[
        \det J > 0.
        \]
        Where \(J\) is the Jacobian matrix \cref{def:jacobian-matrix}.\\
        If the manifold has a maximal compatible atlas, then we say the manifold is 
        orientable, and we may call its corresponding 
        orientation positive and denote the atlas \(\Phi_+\).
    \end{definition}
    \begin{theorem}
        A manifold has either no orientation (any atlas is not compatible) or 
        two orientations.
    \end{theorem}
    \begin{theorem}[Orientability and Existence of Forms of Highest Degree]\label{thm:volume-form-orientability}
        A manifold is orientable iff there exists a nowhere vanishing differential 
        form of the highest degree.
    \end{theorem}
    \subsubsection{Positively Oriented Boundary}
    \begin{definition}[Positively Oriented Boundary]\label{def:positive-boundary}
        Let \(\manifold[M]\) be a orientable \(C^\infty\) manifold with dimension 
        \(m\), positively oriented by compatible atlas \(\Phi_+\). Define coordinate 
        charts on \(\partial \manifold[M]\) from \(\Phi\) as follows, 
        \[
        \phi^{\partial \manifold[M]} : U_\phi \cap \partial \manifold[M] 
        \to \mathbb{R}^{m-1}, 
        \]
        Then \(\Phi^{\partial \manifold[M]}_+ := 
        \Set{\phi^{\partial \manifold[M]}| \phi \in \Phi_+}\) determines an 
        orientation on \(\partial \manifold[M]\), called the positive orientation.
    \end{definition}

    \subsection{Pseudoforms (Densities)}
    \subsubsection{Pseudoscalars}
    \begin{definition}[Space of Pseudoscalars]\label{def:pseudoscalar-space}
        Given a manifold \(\manifold[M]\) and a point \(p\) on it.
        Choose an arbitrary orientation \(o_p=\Phi_+\) of \(T_p\manifold[M]\). On the 
        set \(\mathbb{R}\times \{o_p, -o_p\}\), define an equivalence relation,
        \[
        {\equiv} : (c, o_p) \equiv (-c, -o_p).
        \]
        We denote \(\pseudoscalar[p] := \mathbb{R}\times \{o_p, -o_p\} / {\equiv}\), 
        and call it the space of pseudoscalars. Under a choice of local orientation 
        \(o\), we often denote its element \((c, o) \equiv (-c, -o)\) by 
        \(\pseudo{c}_o\).
    \end{definition}
    \newpage
    \begin{theorem}[Pseudoscalar Vector Space]\label{thm:pseudoscalar-vector-space}
        The space \(\pseudoscalar[p]\) forms a vector space under the operations,
        \[
        \begin{aligned}
            &(a, o) + (b, o) = (a+b, o), \\
            &c(a, o) = (ca, o).
        \end{aligned}
        \]
        Specifically, \(\dim \pseudoscalar[p]=1\), since \(\forall \pseudo{c}_o \in 
        \pseudoscalar[p]\), \(\exists~c \in \mathbb{R}\) s.t. 
        \(\pseudo{c}_o = c\pseudo{1}_o\).
    \end{theorem}
    \begin{theorem}[Pseudoscalar Vector Bundle]\label{thm:pseudoscalar-vector-bundle}
        On the space \(\pseudoscalar = \bigcup_{p \in \manifold[M]}^{}
        \pseudoscalar[p]\), define the canonical projection \(\pi : \pseudoscalar \to 
        \manifold[M]\) by \(\pseudoscalar[p] \mapsto p\).\\
        For any chart \(\phi : U \to \mathbb{R}^m \in \Phi\) of the underlying 
        manifold \(\manifold[M]\), it induces an orientation on \(T_p\manifold[M]\), 
        denoted by \(o_\phi\). We define the local trivialization by 
        \[
        \begin{aligned}
            \tilde{\phi} : \tilde{\pi}(U) &\to U \times \mathbb{R} \\
            (p, (c, o_\phi)) &\mapsto (p, c).
        \end{aligned}
        \]
        And this makes pseudoscalars a vector bundle on any manifold \(\manifold[M]\).
    \end{theorem}
    \begin{theorem}[Pseudoscalars Under Coordinate Transformation]\label{thm:pseudoscalar-coordinate-transformation}
        Given two charts \(\phi : U \to \mathbb{R}^m, \psi : V \to \mathbb{R}^m \in 
        \Phi\) of a manifold \(\manifold[M]\) and a point \(p \in U \cap V\), the 
        following relationship holds on the pseudoscalar space \(\pseudoscalar[p]\)
        \cref{def:pseudoscalar-space},
        \[
        \pseudo{c}_\phi=(\sgn \det J)\pseudo{c}_\psi,
        \]
        where \(\pseudo{c}_\phi\) means using the orientation induced by \(\phi\), 
        and \(\sgn \det J\) is the sign of the Jacobian matrix 
        \cref{def:jacobian-matrix}.
    \end{theorem}
    \begin{proof}
        If \(o_\phi = o_\psi\), that is, \(\det J > 0\) and their orientations agree, 
        then \((c, o_\phi)\equiv(c, o_\psi)\).
        But if \(o_\phi = -o_\psi\), that is, \(\det J < 0\) and their orientations 
        disagree, then \((c, o_\phi)\equiv(-c, o_\psi)\).\par
        This shows that pseudoscalars "flip sign" under charts of different orientation.
    \end{proof}

    \subsubsection{Definition}
    \begin{definition}[Pseudoforms]\label{def:pseudoforms}
        The space of pseudoforms \(\pseudoform{k}{M}\) is defined by the space of 
        differential forms tensored with (twisted by) the pseudoscalar bundle,
        \[
        \pseudoform{k}{M} = \bigwedge^kT^*\manifold[M] \otimes 
        \pseudoscalar(\manifold[M]).
        \]
        This means, pseudoforms are defined locally as 
        \[
        \pseudo{\omega} = \pseudo{1}_\phi\omega\indices{_{i_1}_{\dots}_{i_k}}
        dx^{i_1}\wedge\dots\wedge dx^{i_k}.
        \]
    \end{definition}
    \subsubsection{Pseudoforms and Forms on Orientable Manifolds}
    \begin{theorem}[Conversion of Pseudoforms and Forms]\label{thm:pseudo-to-form}
        For a pseudoform \(\pseudo{\omega} \in \pseudoform{k}{M}\) 
        on an orientable manifold \(\manifold[M]\) of degree \(m\) positively oriented 
        by the atlas \(\Phi_+\), the positive orientation is chosen in a continuous 
        manner. Then \(\pseudoscalar(\manifold[M])\) is isomorphic to \(\mathbb{R}\) 
        by the positive orientation in a continuous manner, so we can say
        \[
        \begin{aligned}
            \iota : \pseudoform{k}{M} &\to \form{k}{M} \\
            \pseudo{1}_\phi\omega\indices{_{i_1}_{\dots}_{i_k}}dx^{i_1}\wedge 
            \dots\wedge dx^{i_k} &\mapsto o(\phi)\omega\indices{_{i_1}_{\dots}_{i_k}}
            dx^{i_1}\wedge\dots\wedge dx^{i_k}
        \end{aligned}
        \]
        defines a smooth differential form on \(\manifold[M]\). Moreover, we can also 
        convert it back, 
        \[
        \begin{aligned}
            \iota ^{-1} : \form{k}{M} &\to  \pseudoform{k}{M}\\
            \omega\indices{_{i_1}_{\dots}_{i_k}}dx^{i_1}\wedge 
            \dots\wedge dx^{i_k} &\mapsto \pseudo{1}_\phi o(\phi)\omega\indices{_{i_1}_{\dots}_{i_k}}
            dx^{i_1}\wedge\dots\wedge dx^{i_k}.
        \end{aligned}
        \]
    \end{theorem}
    \subsubsection{On Absolute Values}
    \begin{remark}
        Some people would say for \(m=\dim \manifold[M]\), pseudo-\(m\)-forms are 
        absolute values of \(m\)-forms. This literal definition is doable in our 
        formulation, and is given by the following theorem.
    \end{remark}
    \newpage
    \begin{theorem}
        Given vectors \(v_1, \dots, v_m \in T_p\manifold[M]\), denote their 
        orientation by \(o_v\). Let \(\tilde{o}_v(\pseudo{1}_\phi) : \pseudoscalar[p] 
        \to \mathbb{R}\) by \((c, o_v) \mapsto c\).
        \[
        \begin{aligned}
            &\left(\tilde{o}_v(\pseudo{1}_\phi)f(x)dx^1\wedge\dots\wedge dx^m\right)
            (v_1, \dots, v_m) \\
            =&\lvert\left( f(x)dx^1\wedge\dots\wedge dx^m \right)
            (v_1, \dots, v_m)\rvert .
        \end{aligned}
        \]
    \end{theorem}
    \begin{remark}
        To understand these in a deeper level, we should think about why we cannot 
        have a nowhere-vanishing differential form on a non-orientable manifold. 
        Let's try to define a usual form on the M{\"o}bius band.\par
        \begin{center}
            \includegraphics[width=\textwidth]{./integral/figs/moebius_chart.pdf}
        \end{center}
        We create a M{\"o}bius band by patching three charts together, with 
        overlapping areas indicated by the same color. \(V_3 \cap V_1\) are patched 
        together twisted, indicated by the reverse direction of arrows.\par
        Let's say we have the three forms
        \[
        \begin{aligned}
            \omega_1 &= dx_1^1\wedge dx_1^2 \\
            \omega_2 &= f_2dx_2^1\wedge dx_2^2 \\
            \omega_3 &= f_3dx_3^1\wedge dx_3^2
        \end{aligned}
        \]
        defined on \(V_1, V_2, V_3\), respectively, and with \(\omega_1\) prescribed 
        to be 1 everywhere. Then by the pullback criterion, 
        \cref{def:manifold-form}, we have 
        \[
        \begin{aligned}
            \omega_1 &= (\phi_2 \circ \phi_1 ^{-1})^*\omega_2 \\
                     &= f_2(\phi_2 \circ \phi_1 ^{-1}(x_1)) dx_1^1\wedge dx_1^2 \\
            \omega_2 &= (\phi_3 \circ \phi_2 ^{-1})^*\omega_3 \\
                     &= f_3(\phi_3 \circ \phi_2 ^{-1}(x_2)) dx_2^1\wedge dx_2^2 \\
            \omega_3 &= (\phi_1 \circ \phi_3 ^{-1})^*\omega_1 \\
                     &= -f_1(\phi_1 \circ \phi_3 ^{-1}(x_1)) dx_3^1\wedge dx_3^2. \\
        \end{aligned}
        \]
        But this cannot be, since by evaluating backwards from the last 
        equation, we have \(f_3 = -1\), \(f_2 = f_3 = -1\), and \(f_1 = f_2 = -1\). 
        This contradicts with the given fact that \(f_1=1\). By this reasoning, we 
        have found an example of the fact that one cannot have a nowhere-vanishing 
        usual form on a non-orientable manifold. Schematically, it looks like this,
        \begin{center}
            \includegraphics[width=\textwidth]{./integral/figs/moebius_form.pdf}
        \end{center}
        where the black line resembles the value of \(\omega\). As you can see, it has 
        a mismatch of sign, making it discontinuous. It is possible to resolve this 
        situation by taking the absolute value, but if so, the linearity is broken. 
        Introducing the pseudoscalar allows us to say \(\pseudo{1}_\phi = 
        -\pseudo{1}_\psi\) if their orientations are reversed, so that we can sort of 
        "fold" the ends together, making it continuous again. In terms of coordinates, 
        \[
        \begin{aligned}
            \pseudo{\omega}_3 &= (\phi_1 \circ \phi_3 ^{-1})^*\pseudo{\omega}_1 \\
            &= -\pseudo{1}_{\phi_1}f_1(\phi_1\circ \phi_3 ^{-1}(x_3))dx_3^1 \wedge 
            dx_3^2 \\
            &= \pseudo{1}_{\phi_3}f_1(\phi_1\circ \phi_3 ^{-1}(x_3))dx_3^1\wedge
            dx_3^2 \\
            &= \pseudo{1}_{\phi_3}dx_3^1\wedge dx_3^2.
        \end{aligned}
        \]
    \end{remark}
    \subsubsection{Metric Volume Form}
    \begin{definition}[The Metric Volume Form]\label{def:metric-volume-form}
        On a smooth Riemannian manifold \(\manifold[M]\) of dimension \(m\) with 
        metric \(g\), the metric volume form is the unique pseudo-\(m\)-form that 
        gives \(\pseudo{1}_\phi\) if the vectors are orthonormal and fits the local 
        orientation of \(\phi\). Especially in 3-dimensional spaces, we denote it 
        as \(\vol\).
    \end{definition}
    \begin{theorem}[Local Representation of Metric Volume Form]\label{thm:local-rep-metric-volume-form}
        On a smooth Riemannian manifold \(\manifold[M]\) of dimension \(m\) with 
        metric \(g\) and coordinate chart \(\phi=x^1, \dots, x^m\), the metric volume 
        form has local representation
        \[
        \vol = \pseudo{1}_\phi \sqrt{\det g}\ dx^1\wedge \dots\wedge dx^m,
        \]
        where \(\det g\) means \(\det g_{ij}\), the determinant of the matrix of 
        metric tensor.
    \end{theorem}
    \begin{proof}
        Let \(\vol = \pseudo{1}_\phi dx^1\wedge\dots\wedge dx^m\) with the tangent 
        vectors \(\tvec{1}{p}, \dots, \tvec{m}{p}\) orthonormal in the sense of \(g\).
        Then \(g\indices{_\mu_\nu}=\delta\indices{_\mu_\nu}\) in that basis.\par
        Under another coordinates \(\psi=(y^1, \dots, y^m)\), we have 
        \[
        \begin{aligned}
            \vol &= \pseudo{1}_\phi \det \left(\frac{\partial x}{\partial y}\right)
            dy^1\wedge\dots\wedge dy^m \\
            &= \pseudo{1}_\psi \left|\det \left(\frac{\partial x}{\partial y}\right)\right|
            dy^1\wedge\dots\wedge dy^m.
        \end{aligned}
        \]
        But 
        \[
        \begin{aligned}
            g &= \delta\indices{_i_j}dx^i\otimes dx^j \\
            &= \delta\indices{_i_j}\frac{\partial x^i}{\partial y^\alpha}
            \frac{\partial x^j}{\partial y^\beta}dy^\alpha\otimes dy^\beta
        \end{aligned}
        \]
        by \cref{thm:tensor-transformation}. So 
        \[
        \begin{aligned}
            g\indices{_\alpha_\beta}&=\sum_{k} \frac{\partial x^k}{\partial y^\alpha}
            \frac{\partial x^k}{\partial y^\beta} \\
            &=\sum_{k} (J ^{-1})_{k\alpha}(J ^{-1})_{k\beta} \\
            &=\sum_{k} ((J ^{-1}) ^{\top})_{\alpha k}(J ^{-1})_{k\beta} \\
            &=\left((J ^{-1}) ^{\top}J ^{-1}\right)_{\alpha\beta}.
        \end{aligned}
        \]
        Thus 
        \[
        \det g\indices{_\alpha_\beta} = \det \left((J ^{-1}) ^{\top}J ^{-1}\right)
        =\left(\det (J ^{-1})\right)^2 = \left|\det \frac{\partial x}{\partial y}
        \right|^2
        \]
        and the proof is completed.
    \end{proof}
    \subsection{Integration of Forms of Highest Degree}
    \begin{definition}[Integration]\label{def:top-form-integration}
        Let \(\manifold[M]\) be a paracompact \(C^\infty\) manifold of dimension 
        \(m\). 
        Choose a \(C^\infty\) partition of unity \(\rho_j, j \in J\) of 
        \(\manifold[M]\) s.t. \(\supp \rho_j \subseteq U_{\phi_j} := U_j\). \\
        Let a pseudo-\(m\)-form \(\pseudo{\omega} \in 
        \pseudoform{m}{M}\) has local expression 
        \(\tilde{\omega}_{\phi_j} = \pseudo{1}_{\phi_j}f_j(x)dx^1_j\wedge \dots\wedge dx^m_j\).
        \[
        \int_{\manifold[M]} \tilde{\omega} = 
        \sum_{j \in J}^{} \int_{(-\infty, 0] \times \mathbb{R}^{m-1}}
        (\rho_j \circ \phi_j ^{-1})(x) f_j(x) dx^1\dots dx^m
        \]
        if the finite sum exists and has the same value for all choices of 
        \(\rho_j\) and \(\phi_j\).
    \end{definition}
    \begin{remark}
        The following theorem reveals why we integrate pseudoforms, not usual forms.
    \end{remark}
    \begin{theorem}[Criterion of Existence of Integral]\label{thm:cpt-form-integral-exist}
        If \(\supp \tilde{\omega}\) is compact, then 
        \(\int_{\manifold[M]} \tilde{\omega}\) exists.
    \end{theorem}
    \begin{proof}
        Let two sets of coordinate charts be 
        \[
        \begin{aligned}
            \phi_j &: U_j \to V_j, j \in J \\
            \phi_k'&: U'_k \to V'_k, k \in K.
        \end{aligned}
        \]
        And cooresponding partition of unity be \(\rho_j, \rho'_k\).\par
        (The goal) Show
        \[
        \begin{aligned}
            &\sum_{j \in J}^{} \int
            (\rho_j \circ \phi_j ^{-1})(x) f_j(x) dx^1\dots dx^m \\
            =&\sum_{k \in K}^{} \int
            (\rho'_k \circ {\phi'_k} ^{-1})(x') f'_k(x') {dx'}^1\dots {dx'}^m
        \end{aligned}
        \]
        \par
        (Split using \(\rho_k'\))
        \[
        \begin{aligned}
            \int_{\manifold[M]} \pseudo{\omega} &= 
            \sum_{j \in J}^{} \int
            (\rho_j \circ \phi_j ^{-1})(x) f_j(x) dx^1\dots dx^m \\
            &= \sum_{j \in J}^{} \int \sum_{k \in K}^{}
            (\rho'_k\circ \phi_j ^{-1})(x)(\rho_j\circ \phi_j ^{-1})(x) f_j(x)
            dx^1\dots dx^m.
        \end{aligned}
        \]
        Since the sum is finite, and \(\supp\pseudo{\omega}\) is compact, and therefore 
        the integral is not improper; thus, there can be no limit or Fubini 
        problems on exchanging sums and integrals. So 
        \[
        \begin{aligned}
            \int_{\manifold[M]} \pseudo{\omega}
            &= \sum_{j \in J}^{}\sum_{k \in K}^{} \int 
            (\rho_j\rho'_k\circ \phi_j ^{-1})(x)f_j(x)
            dx^1\dots dx^m \\
        \end{aligned}
        \]
        \par
        (Change of variables)
        First fix \(j, k\).
        \[
        \begin{aligned}
            &\int (\rho_j\rho'_k\circ \phi_j ^{-1})(x)f_j(x) dx^1\dots dx^m \\
            =&\int (\rho_j\rho'_k\circ \phi_j ^{-1})(\phi_j\circ{\phi_k'}^{-1}(x'))
            f_j(\phi_j\circ{\phi_k'}^{-1}(x')) 
            \left|\det \left(\frac{\partial x}{\partial x'}\right)\right|
            {dx'}^1\dots {dx'}^m \\
            =&\int (\rho_j\rho'_k\circ{\phi_k'}^{-1}(x'))
            f_j(\phi_j\circ{\phi_k'}^{-1}(x')) 
            \left|\det \left(\frac{\partial x}{\partial x'}\right)\right|
            {dx'}^1\dots {dx'}^m
        \end{aligned}
        \]\par
        (Pullback Condition) From \cref{def:manifold-form},
        \[
        \pseudo{\omega}_{\phi_k'}=\left(\phi_j\circ {\phi_k'}^{-1}\right)^*
        \pseudo{\omega}_{\phi_j}.
        \]
        Using \cref{thm:pseudoscalar-coordinate-transformation}, we see,
        \[
        \begin{aligned}
            &\pseudo{1}_{\phi_k'}f'_k(x'){dx'}^1\wedge\dots\wedge{dx'}^m \\
            =&\pseudo{1}_{\phi_j}f_j(\phi_j\circ {\phi_k'}^{-1}(x'))
            \left(\frac{\partial x^1}{\partial {x'}^{\ell_1}}{dx'}^{\ell_1}\right)
            \wedge \dots\wedge 
            \left(\frac{\partial x^m}{\partial {x'}^{\ell_m}}{dx'}^{\ell_m}\right) \\
            =&\pseudo{1}_{\phi_j}f_j(\phi_j\circ {\phi_k'}^{-1}(x'))
            \sum_{\sigma \in S_m}^{}
            (\sgn \sigma)\frac{\partial x^1}{\partial {x'}^{\sigma(1)}}\dots
            \frac{\partial x^m}{\partial {x'}^{\sigma(m)}}
            {dx'}^1\wedge\dots\wedge{dx'}^m \\
            =&(\sgn\det J)\pseudo{1}_{\phi_k'}f_j(\phi_j\circ {\phi'_k}^{-1}(x'))
            \det \left(\frac{\partial x}{\partial x'}\right)
            {dx'}^1\wedge\dots\wedge{dx'}^m \\
            =&\pseudo{1}_{\phi_k'}f_j(\phi_j\circ {\phi'_k}^{-1}(x'))
            \left|\det \left(\frac{\partial x}{\partial x'}\right)\right|
            {dx'}^1\wedge\dots\wedge{dx'}^m.
        \end{aligned}
        \]
        So we see 
        \[
        f'_k(x')=f_j(\phi_j\circ {\phi'_k}^{-1}(x'))
        \left|\det \left(\frac{\partial x}{\partial x'}\right)\right|.
        \]
        Therefore, the integral 
        \[
        \begin{aligned}
            &\int (\rho_j\rho'_k\circ{\phi_k'}^{-1}(x'))
            f_j(\phi_j\circ{\phi_k'}^{-1}(x')) 
            \left|\det \left(\frac{\partial x}{\partial x'}\right)\right|
            {dx'}^1\dots {dx'}^m \\
            =&\int (\rho_j\rho'_k\circ{\phi_k'}^{-1}(x'))
            f'_k(x')
            {dx'}^1\dots {dx'}^m
        \end{aligned}
        \]\par
        (Closing) By moving the sum wrt \(j \in J\) into the integral and using 
        the property of partition of unity, the proof is completed.
    \end{proof}
    \subsection{Integration of Forms of Lower Degree}
    \subsubsection{Pseudoorientation (Transverse Orientation)}
    \begin{remark}
        Because pseudoscalars depend on local orientations, we need an extra 
        structure in order to pullback pseudoforms onto a submanifold.
    \end{remark}
    \begin{definition}[Pseudoorientation]\label{def:pseudoorientation}
        For \(\manifold[Z]\) a submanifold of \(\manifold[M]\), pseudoorientation is 
        a map that for all \(p \in \manifold[Z]\), takes a local orientation of 
        \(\manifold[Z]\) at \(p\) (an orientation of \(T_p\manifold[Z]\)) to a local 
        orientation of \(\manifold[M]\) at \(p\) (an orientation of 
        \(T_p\manifold[M]\)) 
        \textbf{continuously in \(p\), and takes opposite orientations 
        to opposite orientations}.
    \end{definition}
    \begin{remark}
        Requiring well-defined surface normal (with a metric, of course) is precisely 
        this criterion for \((m-1)\) dimensional submanifolds, for we can form an 
        orientation of \(\manifold[M]\) from an orientation of \(\manifold[Z]\) (which 
        is essentially \((m-1)\) vectors on \(T_p\manifold[Z]\)) and the normal.\\
        This makes \(\textnormal{M{\"o}}\) integrable as a submanifold of 
        \(\textnormal{M{\"o}}\times \mathbb{R}\), since we can choose the surface normal 
        to be positive everywhere. But \(\textnormal{M{\"o}}\) is not integrable as a 
        submanifold of \(\mathbb{R}^3\), since if we try to assign surface normal 
        along the strip, when we get back to where we started, the direction would be 
        reversed.
    \end{remark}
    \newpage
    \begin{definition}[Pullback of Pseudoscalars]\label{def:pullback-pseudoscalar}
        Given a manifold \(\manifold[M]\) and its submanifold \(\manifold[Z]\), 
        and \(\psi\) be a chart of \(\manifold[M]\), \(\phi\) a chart of 
        \(\manifold[Z]\). At the point \(p\), if the pseudoorientation \(i\)
        \cref{def:pseudoorientation} is chosen such that the local orientations 
        \(o_\phi \mapsto o_\psi\), then we say \(i^*\pseudo{1}_\psi = 
        \pseudo{1}_\phi\). If \(o_\phi \mapsto -o_\psi\), then \(i^*\pseudo{1}_\psi = 
        -\pseudo{1}_\phi\).
    \end{definition}
    \subsubsection{Definition}
    \begin{definition}[Integration of Lower Degree Forms]\label{def:pullback-integral}
        Let \(\manifold[Z]\) be an pseudooriented \(C^\infty\) submanifold of 
        dimension \(d\) of \(\manifold[M]\), \(i : \manifold[Z] \to \manifold[M]\) be 
        the immersion map. \\
        Let \(\omega \in \form{d}{M}\), we define 
        \[
        \int_{\manifold[Z]}\omega := \int_{\manifold[Z]}^{} i^*\omega
        \]
        if it exists.
    \end{definition}
    \subsubsection{Stoke's theorem}
    \begin{theorem}[Stoke's Theorem]\label{thm:stokes-theorem}
        If \(\manifold[M]\) is an oriented \(C^\infty\) manifold of dimension \(m\) 
        and \(\omega \in \form{{m-1}}{M}\), then 
        \[
        \int_{\manifold[M]}^{}d\omega = \int_{\partial \manifold[M]}^{}\omega
        := \int_{\partial \manifold[M]}^{} i^*\omega,
        \]
        where \(i : \partial \manifold[M] \to \manifold[M]\) is just the immersion 
        map, \(i : p \mapsto p\).
    \end{theorem}
    \begin{proof}
        (Partition Using Charts) Choose a \(C^\infty\) partition of unity 
        \cref{thm:manifold-partition-of-unity} \(\rho_j, j \in J\) s.t. 
        \(\supp \rho_j\) are compact and \(\supp \rho_j \subseteq U_{\phi_j} 
        := U_j\). \par
        Now \(\omega = \sum_{j \in J}^{} \rho_j \omega\) is a finite sum. So 
        it suffices to show that, if \(\eta \in \form{m-1}{M}\), \(\supp \eta\) 
        compact and \(\supp\eta \subseteq U_\phi\) then \(\int_{\manifold[M]}^{}
        d\eta = \int_{\partial \manifold[M]}^{} \eta\), and apply 
        \(\eta = \rho_j\omega\) for all \(j \in J\). \par
        (The integral)
        Suppose the coordinates of \(\phi\) is labeled \(x^1, \dots, x^m\).
        % Since \(\supp \eta \subseteq U_\phi\), we will make no distinction between 
        % \(\eta \in \form{m-1}{M}\) and \(\eta_\phi \in \eucForm{m-1}{m}\). \par
        Locally, let 
        \[
        \eta = \sum_{\ell=1}^{m} f_\ell dx^1\wedge \cdots \cancel{dx^\ell} \cdots 
        \wedge dx^m,
        \]
        where \(f_\ell : (-\infty, 0] \times \mathbb{R}^{m-1} \to \mathbb{R}\), and 
        it is \(C^\infty\). Then, also locally, 
        \[
        \begin{aligned}
            d\eta &= \sum_{\ell = 1}^{m} \frac{\partial f_\ell}{\partial x^\ell} dx^\ell 
            \wedge dx^1\wedge \cdots \cancel{dx^\ell} \cdots \wedge dx^m \\
            &= \left(\sum_{\ell=1}^{m} (-1)^{\ell-1}
            \frac{\partial f_\ell}{\partial x^\ell}\right)dx^1\wedge \dots \wedge dx^m.
        \end{aligned}
        \]
        Then by \cref{def:top-form-integration}, 
        \[
        \int_{\manifold[M]}^{}d\eta = \int_{(-\infty, 0]\times \mathbb{R}^{m-1}}^{} 
        \left(\sum_{\ell=1}^{m} (-1)^{\ell-1}
        \frac{\partial f_\ell}{\partial x^\ell}\right)dx^1\dots dx^m.
        \]\par
        (To be precise, we need to choose another partition of unity \(\rho_j'\) to 
        do integration. But we can just choose it to cover all of \(\supp \eta\) 
        and don't care all other parts, so that doesn't matter too much.)\par
        Choose a rectangular region \(R\) s.t. 
        \[
        \supp \eta \subseteq [a_1, 0]\times \dots \times [a_m, b_m]
        \]
        and define 
        \[
        R_\ell := [a_1, 0] \times \dots \cancel{[a_\ell, b_\ell]} \dots 
        \times [a_m, b_m].
        \]\par
        (\(\ell = 2,\dots, m\)) In this case, by Fubini and FTC, 
        \[
        \begin{aligned}
            &\int_{R}^{}\frac{\partial f_\ell}{\partial x^\ell}dx^1\dots dx^m\\
            =&\int_{R_\ell}^{}\left(\int_{a_\ell}^{b_\ell} 
            \frac{\partial f_\ell}{\partial x^\ell}dx^\ell\right)dx^1\dots
            \cancel{dx^\ell}\dots dx^m \\
            =&\int_{R_\ell}^{}
            (\underbrace{f_\ell(x^1, \dots, b_\ell, \dots, x^m)}_{0}
             - \underbrace{f_\ell(x^1, \dots, a_\ell, \dots, x^m)}_{0})
            dx^1\dots \cancel{dx^\ell}\dots dx^m \\
            =&0, 
        \end{aligned}
        \]
        since \(\supp\eta \subseteq R\), so on the boundary \(f=0\). \par
        (\(\ell = 1\)) Now the integral has only one term left.
        \[
        \begin{aligned}
            \int_{\manifold[M]}^{} d\eta &= 
            \int_{R}^{} \frac{\partial f_1}{\partial x^1}dx^1\dots dx^m \\
            &=\int_{R_1}^{} \left(\int_{a_1}^{0}\frac{\partial f_1}{\partial x^1}
            dx^1\right)dx^2\dots dx^m \\
            &=\int_{R_\ell}^{}
            (f_1(0, x^2, \dots, x^m) - \underbrace{f_1(a_1, x^2, \dots, x^m)}_{0})
            dx^2\dots dx^m \\
            &=\int_{\mathbb{R}^{m-1}}^{} (f_1\circ i) dx^2\dots dx^m \\
            &=\int_{\partial \manifold[M]}^{} i^*\eta.
        \end{aligned}
        \]
    \end{proof}

    \begin{theorem}[Uniqueness of Dual Inner Product]\label{thm:dual-inner-product-uniqueness}
        Let \(\manifold[M]\) be an \(m\)-dimensional Riemannian manifold with metric 
        \(g\). Then for all \(k = 1, \dots, m\), \(g\) determines a unique inner 
        product \(\langle \cdot, \cdot\rangle\) s.t. 
        \[
        \langle \omega^1 \wedge \dots\wedge \omega^k, 
        \eta^1\wedge\dots\wedge\eta^k\rangle = \det \left(\left\langle
            \indUp{(\omega^i)}, \indUp{(\eta^j)}\right\rangle_g\right)
        \]
    \end{theorem}
    \begin{proof}
        (Show \(dx^I\) is an orthonormal basis)
        Let \(I, J\) be increasing \(k\)-indices, and let \(\tfld{1}, \dots, 
        \tfld{k}\) be local orthonormal frame. Thus, \(g\indices{_i_j}=
        \delta\indices{_i_j}\), and \(\indUp{(dx^i)}=\tfld{i}\).\par
        If \(I = J\),
        \[
        \left\langle dx^I, dx^I\right\rangle = \det 
        (\langle \tfld{i}, \tfld{j}\rangle)=1.
        \]
        But if \(I \neq J\), \(\left\langle dx^I, dx^J\right\rangle=0\) since there 
        would be zero rows.\par
        Now define inner product by 
        \[
        \langle\omega, \eta\rangle=\left\langle \omega_Idx^I, \eta_Jdx^J\right\rangle
        :=\omega_I\eta_J \delta\indices{^I^J}.
        \]
        (I cannot show it satisfiy the desired property; but I convinced myself using 
        examples.)\par
        (It remains to show that any other orthonormal frame produces the same 
        inner product. But I hadn't think of a good way.)
    \end{proof}
    \begin{theorem}[Volume Form Is Of Unit Norm]\label{thm:volume-form-unit-norm}
        On a Riemannian manifold \(\manifold[M]\) with dimension \(m\) and metric 
        \(g\), the metric volume form \(\vol\) is the only \(m\)-form that has 
        unit norm with respect to the inner product \cref{thm:dual-inner-product-uniqueness}.
    \end{theorem}
    \begin{proof}
        Let an \(m\)-form \(\omega=\lambda dx^1\wedge\dots\wedge dx^m\). We wish 
        \[
        \begin{aligned}
            1 = \langle\omega, \omega\rangle
            &= \lambda^2\langle dx^1\wedge\dots\wedge dx^m, 
            dx^1\wedge\dots\wedge dx^m\rangle\\
            &= \lambda^2 \det \left(\left<\indUp{(dx^i)}, \indUp{(dx^j)}\right>\right)\\
            &= \lambda^2 \det \left(g_{\mu\nu}{\indUp{(dx^i)}}^\mu{\indUp{(dx^j)}}^\nu\right).
        \end{aligned}
        \]
        Now, since \((dx^i)_\mu=\delta\indices{^i_\mu}\), we know 
        \({\indUp{(dx^i)}}^\mu=g^{\mu\nu}\delta\indices{^i_\nu}=g^{\mu i}\). Moreover, 
        \(g_{\mu\nu}g^{\nu\sigma}=\delta\indices{_\mu^\sigma}\). So 
        \[
        \begin{aligned}
            1 &= \lambda^2 \det \left(g_{\mu\nu}g^{\mu i}g^{\nu j}\right)\\
            &= \lambda^2 \det \left(g^{i\mu}g_{\mu\nu}g^{\nu j}\right)\\
            &= \lambda^2 \det \left(g^{i\mu}\delta\indices{_\mu^j}\right)\\
            &= \lambda^2 \det \left(g^{ij}\right).
        \end{aligned}
        \]
        Therefore, \(\lambda=\pm\sqrt{\det \left(g_{ij}\right)}\).
    \end{proof}

    \subsubsection{Definition and Uniqueness}
    \begin{theorem}[Defining Property of Hodge Star]\label{thm:hodge-star}
        On a Riemannian manifold \(\manifold[M]\) with dimension \(m\) and metric 
        \(g\), there exists a unique bundle homomorphism \(\hodge : \form{k}{M} \to 
        \pseudoform{m-k}{M}\) s.t. 
        \[
        \omega\wedge \hodge \eta=\langle \omega, \eta\rangle_g \vol.
        \]
    \end{theorem}
    \begin{proof}
        (uniqueness) Suppose there exist two such homomorphisms, \(\hodge\) and 
        \(\hodge'\). Then for all \(\omega \in \form{k}{M}\),
        \[
        \begin{aligned}
            \omega\wedge \hodge \eta&=\langle \omega, \eta\rangle_g \vol\\
            \omega\wedge \hodge' \eta&=\langle \omega, \eta\rangle_g \vol
        \end{aligned}
        \]
        so \(\omega\wedge (\hodge\eta-\hodge'\eta)=0\). Thus 
        \(\hodge\eta=\hodge'\eta\) for all \(\eta \in \form{k}{M}\).\par
        (existence) We claim for a local orthonormal frame \(\tfld{i}\), 
        \[
        \hodge \left(dx^{i_1}\wedge \dots\wedge dx^{i_k}\right)
        =\epsilon\indices{^{i_1}^{\dots}^{i_k}_{j_{k+1}}_{\dots}_{j_m}}
        dx^{j_{k+1}}\wedge\dots\wedge dx^{j_m}.
        \]
        Since 
        \[
        \begin{aligned}
            &\left(dx^{i_1}\wedge \dots\wedge dx^{i_k}\right)\wedge 
            \hodge \left(dx^{i_1}\wedge \dots\wedge dx^{i_k}\right)\\
            =& \left<dx^I, dx^I\right>\vol\\
            =& \vol,
        \end{aligned}
        \]
        we need the Levi-Civita symbol to make the indices in correct order.
    \end{proof}
    \subsubsection{Coordinate Representation of Hodge Dual}
    \begin{theorem}[Hodge Dual of Any Basis]\label{thm:hodge-dual-basis}
        On a Riemannian manifold \(\manifold[M]\) with dimension \(m\) and metric 
        \(g\), the Hodge dual satisfies 
        \[
        \begin{aligned}
            &\hodge \left(dx^{i_1}\wedge \dots\wedge dx^{i_k}\right)\\
            =&\frac{\sqrt{\det g}}{(n-k)!}g^{i_1j_1}\dots g^{i_kj_k}
            \epsilon_{j_1\dots j_k j_{k+1}\dots j_m} dx^{j_{k+1}}\wedge\dots\wedge
            dx^{j_m}.
        \end{aligned}
        \]
        The term \((n-k)!\) is there because we didn't restrict \(j_{k+1}, \dots, 
        j_m\) are in order. If we ensure that, then \((n-k)!\) is not there.
    \end{theorem}
    \begin{proof}
        We verify it by calculating 
        \[
        \left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right) \wedge 
        \left(\frac{\sqrt{\det g}}{(n-k)!}g^{i_1j_1}\dots g^{i_kj_k}
        \epsilon_{j_1\dots j_k j_{k+1}\dots j_m} dx^{j_{k+1}}\wedge\dots\wedge
        dx^{j_m}\right).
        \]
        Firstly, 
        \[
        \begin{aligned}
            \left<dx^{i_1}\wedge\dots\wedge dx^{i_k}, 
            dx^{i_1}\wedge\dots\wedge dx^{i_k}\right>
            &=\det \left(\left<\indUp{(dx^i)}, \indUp{(dx^j)}\right>\right)\\
            &=\det \left(g_{\mu\nu}g^{i\mu}g^{j\nu}\right)\\
            &=\det \left(g^{i\mu}g_{\mu\nu}g^{\nu j}\right)\\
            &=\det \left(g^{ij}\right),
        \end{aligned}
        \]
        but only \(i_1, \dots, i_k\) rows and columns are picked out.\par
        However, \(g^{i_1j_1}\dots g^{i_kj_k}\epsilon_{j_1\dots j_n}\) are precisely 
        this expression, because \(j_{k+1}, \dots, j_m\) cannot contain terms of 
        \(i_1, \dots, i_k\), otherwise they wedge to be \(0\). Therefore, 
        \(j_1, \dots, j_k\) are permutations of \(i_1, \dots, i_k\), and the 
        Levi-Civita symbol takes care of the \((\sgn \sigma)\). Thus the proof is 
        completed.
    \end{proof}
    \begin{corollary}[Coordinate Representation of Hodge Dual]\label{cor:hodge-dual-coordinates}
        Given a \(k\)-form \(\omega \in \form{k}{M}\) on a Riemannian manifold 
        \(\manifold[M]\) with dimension \(m\) and metric \(g\), we have 
        \[
        (\hodge\omega)_{j_{k+1}\dots j_m} = 
        \frac{\sqrt{\det g}}{(n-k)!}g^{i_1j_1}\dots g^{i_kj_k}
        \epsilon_{j_1\dots j_k j_{k+1}\dots j_m} \omega_{i_1\dots i_k}.
        \]
    \end{corollary}
    \subsubsection{Properties of Hodge Dual}
    \begin{theorem}[Double Hodge Dual]\label{thm:double-hodge-dual}
        For any \(\omega \in \form{k}{M}\) on a Riemannian manifold \(\manifold[M]\),
        \[
        \hodge\hodge\omega=(-1)^{k(m-k)}\omega.
        \]
    \end{theorem}
    \begin{remark}
        Note this applies to Euclidean metric only. In other cases, the negative sign 
        may be different. For example, under the Lorentzian metric 
        \(g=({-}{+}{+}{+})\), the expression is instead 
        \[
        \hodge\hodge\omega=(-1)^{k(m-k)+1}\omega.
        \]
        Also, be very careful about the \(k\) when dealing with consecutive Hodges.
    \end{remark}
    \begin{proof}
        It suffices to show for orthonormal frames, 
        \[
        \hodge\hodge \left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right)
        =(-1)^{k(n-k)}\left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right).
        \]
        To show this, we verify according to \cref{thm:hodge-star},
        \[
        \hodge \left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right)
        \wedge (-1)^{k(n-k)}\left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right).
        \]
        We want this to be equal to \(\left<\hodge dx^I, \hodge dx^I\right>\vol\). 
        By the exchanging rule \cref{thm:skew-symmetry-exterior-product}, we have 
        \[
        \begin{aligned}
            =&\left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right)\wedge
            \hodge\left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right)\\
            =&\left<dx^I, dx^I\right>\vol.
        \end{aligned}
        \]
        Thus, if \(\left<dx^I, dx^I\right>=\left<\hodge dx^I, \hodge dx^I\right>\), 
        then we are done. I can prove the case when they are orthonormal, 
        \[
        \begin{aligned}
            &\left<\hodge \left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right), 
            \hodge \left(dx^{i_1}\wedge\dots\wedge dx^{i_k}\right)\right>\\
            =&\left<\delta^{i_1j_1}\dots\delta^{i_kj_k}\epsilon_{j_1\dots j_n}
            dx^{j_{k+1}}\wedge\dots\wedge dx^{j_m}, 
            \delta^{i_1j_1'}\dots\delta^{i_kj_k'}\epsilon_{j_1'\dots j_n'}
            dx^{j_{k+1}'}\wedge\dots\wedge dx^{j_m'}\right>\\
            =&1
        \end{aligned}
        \]
        because only the terms which \(j_{k+1}, \dots, j_m\) didn't contain 
        \(i_1, \dots, i_k\) survive, and Levi-Civita symbol squared is definitely 1.
    \end{proof}
    \begin{theorem}[Hodge Dual of Constant]\label{thm:hodge-dual-constant}
        On any Riemannian manifold \(\manifold[M]\),
        \[
        \hodge 1=\vol.
        \]
        More generally, for any function \(f \in C^\infty(\manifold[M])\),
        \[
        \hodge f=f\vol.
        \]
    \end{theorem}
    \begin{theorem}[Hodge Dual of Flatted Vector Fields]\label{thm:hodge-dual-vector-field}
        On any Riemannian manifold \(\manifold[M]\), given a vector field \(X\), 
        \[
        \hodge \indDown{X}=X\iprod \vol.
        \]
    \end{theorem}
    \begin{proof}
        Again, we verify \cref{thm:hodge-star}. In local coordinates, 
        \[
        \begin{aligned}
            \vol &= \sqrt{\det g}dx^1\wedge\dots\wedge dx^m\\
            X &= X^\mu\tfld{\mu}\\
            \indDown{X}&=\left(\indDown{X}\right)_\mu dx^\mu=g_{\mu\nu}X^\nu dx^\mu.
        \end{aligned}
        \]
        Then, using \cref{thm:iprod-cofactor}, 
        \[
        \begin{aligned}
            &\left(\indDown{X}\right)\wedge \left(X\iprod \vol\right)\\
            =&g_{\mu\nu}X^\nu\sqrt{\det g}\sum_{i=1}^{m} (-1)^{i-1}
            X^i dx^\mu\wedge dx^1\wedge\dots\cancel{dx^i}\dots\wedge dx^m.
        \end{aligned}
        \]
        But the terms that \(\mu \neq i\) vanishes, so 
        \[
        \begin{aligned}
            &\left(\indDown{X}\right)\wedge \left(X\iprod \vol\right)\\
            =&\sqrt{\det g}\sum_{i=1}^{m} (-1)^{i-1}g_{i\nu}X^\nu
            X^i dx^i\wedge dx^1\wedge\dots\cancel{dx^i}\dots\wedge dx^m\\
            =&g_{\mu\nu}X^\mu X^\nu \vol\\
            =&\left<X, X\right>_g\vol.
        \end{aligned}
        \]
    \end{proof}
    \subsubsection{Other Differential Operators}
    \begin{definition}[Codifferential]\label{def:codifferential}
        On a Riemannian manifold \(\manifold[M]\) equipped with metric \(g\), we 
        define the codifferential \(\codiff : \form{k}{M}\to \form{k-1}{M}\).
        \[
        \begin{aligned}
            \codiff &:=(-1)^k\hodge ^{-1} d\hodge\\
            &=(-1)^{m(k+1)+1}\hodge d\hodge.
        \end{aligned}
        \]
        The second expression is given by \cref{thm:double-hodge-dual}, since the 
        inverse Hodge operator is itself times some negative sign.
    \end{definition}
    \begin{theorem}[Twice Codifferential]\label{thm:double-codifferential}
        \[
        \codiff^2=0.
        \]
    \end{theorem}
    \begin{proof}
        \[
        \begin{aligned}
            \codiff^2&=(-1)^{k-1}\hodge ^{-1} d \hodge (-1)^k \hodge ^{-1} d \hodge \\
            &=-\hodge ^{-1}d^2\hodge =0.
        \end{aligned}
        \]
    \end{proof}
    \begin{theorem}[Codifferential Of Functions]\label{thm:function-codiff}
        For any \(f \in C^\infty(\manifold[M])\), 
        \[
        \codiff f = 0.
        \]
    \end{theorem}
    \begin{proof}
        \[
        \codiff f = (-1)^0\hodge ^{-1}\overbrace{d\underbrace{\hodge f}_{f\vol}}
        ^{0}=0
        \]
    \end{proof}
    \begin{definition}[Divergence]\label{def:divergence-vector-field}
        On a Riemannian manifold \(\manifold[M]\) equipped with metric \(g\), we 
        define the divergence of a vector field to be 
        \[
        \begin{aligned}
            \divv X &:=\hodge d \hodge \indDown{X}\\
            &= -\codiff \indDown{X}.
        \end{aligned}
        \]
        The second expression is given by \cref{def:codifferential} and the fact 
        that \(k=1\).
    \end{definition}
    \begin{theorem}[]\label{thm:divergence-vector-field-volume-form}
        The divergence of a vector field \(\divv X\) is the unique scalar that 
        \[
        (\divv X)\vol = d\hodge \indDown{X}.
        \]
    \end{theorem}
    \begin{proof}
        Note that 
        \[
        \divv X = -\codiff \indDown{X} = \hodge ^{-1} d \hodge \indDown{X}.
        \]
        Therefore, 
        \[
        \divv X \wedge \hodge \divv X = \left<\divv X, \divv X\right>\vol.
        \]
        But since \(\divv X\) is just a scalar, the above is equivalent to 
        \[
        (\divv X)\hodge \divv X = (\divv X)^2\vol.
        \]
        If \(\divv X \neq 0\), then 
        \[
        \hodge \divv X = d\hodge \indDown{X} = (\divv X)\vol.
        \]
        If \(\divv X=0\), then by definition \(d\hodge \indDown{X}=0\), so the 
        equation trivially holds.
    \end{proof}
    \begin{theorem}[Divergence In Local Coordinates]\label{thm:divergence-coordinates}
        On a Riemannian manifold \(\manifold[M]\) equipped with metric \(g\), the 
        divergence \cref{def:divergence-vector-field} of a vector field \(X\) has 
        local expression
        \[
        \divv X = \frac{1}{\sqrt{\det g}}\tfld{\mu}\left(X^\mu\sqrt{\det g}\right).
        \]
    \end{theorem}
    \begin{proof}
        Using \cref{thm:hodge-dual-vector-field} and \cref{thm:iprod-cofactor}, 
        \[
        \begin{aligned}
            \divv X &= \hodge d \hodge \indDown{X} = \hodge d(X \iprod \vol)\\
            &=\hodge d \left(\sum_{i=1}^{m} (-1)^{i-1}\sqrt{\det g}X^i
            dx^i\wedge \dots \cancel{dx^i}\dots \wedge dx^m\right)\\
            &=\hodge \sum_{i=1}^{m} (-1)^{i-1}\tfld{\mu} \left(\sqrt{\det g}X^i\right)
            dx^\mu\wedge dx^i\wedge \dots \cancel{dx^i}\dots \wedge dx^m.
        \end{aligned}
        \]
        Now only the \(\mu=i\) terms remain, and since \(\hodge\hodge1=(-1)^{0\cdot
        (m-0)}=1\), \(\hodge \vol=1\), and so 
        \[
        \begin{aligned}
            \divv X &= \hodge \left(\tfld{\nu}\left(\sqrt{\det g}X^\mu\right)
            \frac{1}{\sqrt{\det g}}\vol\right)\\
            &=\frac{1}{\sqrt{\det g}}\tfld{\nu}\left(\sqrt{\det g}X^\nu\right).
        \end{aligned}
        \]
    \end{proof}
    \begin{definition}[Laplace-de Rham Operator]\label{def:LdR-laplace-de-rham-operator}
        On a Riemannian manifold \(\manifold[M]\) equipped with metric \(g\), we 
        define the Laplace-de Rham operator to be 
        \[
        \LdR := -(d\codiff + \codiff d),
        \]
        where \(\codiff\) is the codifferential \cref{def:codifferential}.
    \end{definition}
    \begin{remark}
        Both sign conventions are popular. This particular convention makes the 
        local expression in Euclidean space with the standard metric \(g\) reduce 
        to the familiar "Laplacian". In Minkowski space, the LdR operator is the 
        wave operator \(\dAlembert\), the d'Alembertian.
    \end{remark}
    \begin{theorem}[LdR Operator In Local Coordinates]\label{thm:ldr-coordinates}
        On a Riemannian manifold \(\manifold[M]\) equipped with metric \(g\), the 
        LdR operator \cref{def:LdR-laplace-de-rham-operator} of a function \(f\)
        has local expression 
        \[
        \begin{aligned}
            \LdR f&=-\codiff df=\divv \grad f\\
            &=\frac{1}{\sqrt{\det g}}\tfld{\mu}\left(\sqrt{\det g}\;g^{\mu\nu}
            \tfld{\nu}f\right).
        \end{aligned}
        \]
    \end{theorem}

    \subsubsection{Surface Integrals}
    \begin{remark}
        In vector calculus, we define the integration of a vector field \(X\) over a 
        submanifold \(S\) to be 
        \[
        \int_{\manifold[M]} \langle X, N\rangle_g dS,
        \]
        where \(N\) is the normal vector of \(S\), \(\langle X, N\rangle_g\) is the 
        inner product of the two (induced by metric \(g\)), and \(dS\) is the (metric) 
        volume form of \(S\). We seek to reformulate this in terms of our language.
    \end{remark}
    \begin{theorem}[Hodge Star and Flux]\label{thm:hodge-flux}
        Let \(\manifold[M]\) be an \(m\)-dimensional Riemannian manifold with metric 
        \(g\), and \(\manifold[S]\) be a submanifold of \(\manifold[M]\) of 
        codimension \(1\).
        Let \(X\) be a vector field on \(\manifold[M]\), \(dS\) be the 
        volume forms of \(\manifold[M]\) and \(\manifold[S]\), respectively, 
        and \(i\) be the immersion map. Then 
        \[
        i^*(\hodge \indDown{X})=\langle X, N\rangle_g dS.
        \]
    \end{theorem}
    \begin{proof}
        Let \(X^\perp = \langle X, N\rangle_gN, X^{\top}=X-X^\perp\). Then 
        \[
        X\iprod dV = X^\perp \iprod dV + X ^{\top} \iprod dV.
        \]
        But \(X^\perp = \langle X, N\rangle_gN\), so 
        \[
        X^\perp \iprod dV = \langle X, N\rangle_gN\iprod dV
        = \langle X, N\rangle_g dS.
        \]
        We now show \(X ^{\top} \iprod dV=0\). Consider any \(X_1, \dots, X_{m-1}\) 
        on \(\manifold[S]\), we see 
        \[
        (X ^{\top} \iprod dV)(X_1, \dots, X_{m-1})
        = dV(X^{\top}, X_1, \dots, X_{m-1})=0
        \]
        since \(m\) vectors on \((m-1)\)-dimensional vector spaces are necessarily 
        dependent.
    \end{proof}
    \begin{theorem}[Divergence Theorem]\label{thm:divergence-theorem}
        Let \(\manifold[M]\) be an \(m\)-dimensional Riemannian manifold with metric 
        \(g\). Denote the metric volume form of \(\partial \manifold[M]\) by 
        \(dS\), then for any compactly supported vector field \(X\), we have 
        \[
        \int_{\manifold[M]} (\divv X)\vol=\int_{\partial \manifold[M]} 
        \left<X, N\right>_g dS.
        \]
    \end{theorem}
    \begin{proof}
        By \cref{thm:divergence-vector-field-volume-form}, and Stokes theorem 
        \cref{thm:stokes-theorem}, we see 
        \[
        \int_{\manifold[M]} (\divv X)\vol = \int_{\manifold[M]} d\hodge\indDown{X}
        =\int_{\partial \manifold[M]}i^*\left(\hodge \indDown{X}\right)
        =\int_{\partial \manifold[M]}\left<X, N\right>_g dS.
        \]
    \end{proof}
